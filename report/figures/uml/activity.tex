\begin{tikzpicture}[

    action/.style={draw, rounded corners=5pt, align=center, text width=2.75cm, minimum height=0.62cm,

  \node[font=\footnotesize\bfseries] at (3.3,0.72) {API request (synchronous)};
  \node[font=\footnotesize\bfseries] at (11.4,0.72) {Background job (JobRunner thread)};

  % ---- Lane 1: request handling
  \node[initial] (s) at (3.3,0) {};
  \node[decision] (d1) at (3.3,-0.75) {};
  \node[action] (a1) at (1.6,-1.8) {Parse arXiv identifier};
  \node[action] (u1) at (5.0,-1.95) {Read file (stop above 50\,MB); check PDF signature, encryption, page cap; hash SHA-256};
  \node[decision] (m1) at (3.3,-3.2) {};
  \node[decision] (d2) at (3.3,-4.0) {};
  \node[action] (rej) at (1.5,-5.0) {Return 422 (invalid) or 409 (duplicate) with reason};
  \node[final] (f1) at (1.5,-5.85) {};
  \node[action] (st) at (4.6,-5.0) {Store upload as \emph{sha256}.pdf (uploads only)};
  \node[action] (job) at (4.6,-6.1) {Create queued job, or return the in-flight job for the same source};
  \node[action] (sub) at (4.6,-7.2) {Submit job to JobRunner; respond 202 Accepted};
  \node[bar] (fork) at (4.6,-7.95) {};
  \node[final] (f2) at (4.6,-8.7) {};

  \draw[flow] (s) -- (d1);
  \draw[flow] (d1.west) -| node[g, above, pos=0.3] {[arXiv ID]} (a1.north);

  \draw[flow] (d2.east) -| node[g, above, pos=0.25] {[valid]} (st.north);
  \draw[flow] (st) -- (job);
  \draw[flow] (job) -- (sub);
  \draw[flow] (sub) -- (fork);
  \draw[flow] (fork) -- (f2);

  \draw[dashed, rounded corners=8pt] (7.5,0.2) rectangle (15.3,-9.15);
  \node[action] (p1) at (11.2,-0.45) {state = fetching (attempts + 1)};
  \node[decision] (d3) at (11.2,-1.3) {};
  \node[action] (a2) at (9.2,-2.3) {Fetch arXiv metadata};
  \node[decision] (d4) at (9.2,-3.25) {};
  \node[action, text width=2.0cm] (a3) at (11.75,-3.25) {state = ready (version already indexed)};

  \node[action, text width=2.2cm] (u2) at (13.9,-2.3) {Read stored upload};

  \node[action] (e1) at (11.2,-5.85) {state = extracting: text per page and page offsets (PyMuPDF)};

  \node[action] (e3) at (11.2,-7.7) {state = embedding: MiniLM, batched};

  \node[final] (f4) at (11.2,-10.6) {};

  \draw[flow] (p1) -- (d3);
